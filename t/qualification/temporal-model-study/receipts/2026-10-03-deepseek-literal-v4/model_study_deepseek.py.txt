"""Direct configured DeepSeek SSE inference; credential stays in the worker.

The parent enforces five seconds without real text/thinking deltas, including
connection startup. Pings, headers and usage cannot keep a stalled call alive.
Thinking text is not persisted; only its byte length/digest is retained.
"""
import hashlib
import json
import os
from pathlib import Path
import selectors
import shlex
import signal
import subprocess
import sys
import time
from urllib.parse import urlsplit
from model_study_transport import ModelTransportError, strict_json


def configuration(path):
    values={}
    for line in Path(path).expanduser().read_text().splitlines():
        line=line.strip()
        if not line or line.startswith('#'):continue
        if line.startswith('export '):line=line[7:]
        name,sep,value=line.partition('=')
        if not sep:raise ValueError('invalid environment assignment')
        if name.strip() not in ('DEEPSEEK_API_KEY','ANTHROPIC_API_KEY','ANTHROPIC_AUTH_TOKEN',
                'ANTHROPIC_BASE_URL','ANTHROPIC_MODEL','CLAUDE_CODE_EFFORT_LEVEL'):continue
        parts=shlex.split(value,comments=True)
        if len(parts)>1:raise ValueError('invalid environment value')
        values[name.strip()]=parts[0] if parts else ''
    if not values.get('DEEPSEEK_API_KEY'):raise ValueError('DeepSeek credential is missing')
    base=values.get('ANTHROPIC_BASE_URL','').rstrip('/')
    parsed=urlsplit(base)
    if parsed.scheme!='https' or parsed.hostname!='api.deepseek.com' or parsed.path!='/anthropic':
        raise ValueError('expected configured official DeepSeek Anthropic endpoint')
    model=values.get('ANTHROPIC_MODEL')
    effort=values.get('CLAUDE_CODE_EFFORT_LEVEL')
    if not model or effort not in ('low','medium','high','max','none'):raise ValueError('configured model/effort missing or unsupported')
    return values,dict(model=model,modelProvider='deepseek',reasoningEffort=effort,endpoint=base+'/v1/messages')


def output(value):print(json.dumps(value,ensure_ascii=False),flush=True)


def worker(request):
    import httpx
    values,metadata=configuration(request['env_file'])
    body=dict(model=metadata['model'],max_tokens=32768,stream=True,
        system='Answer only from the supplied task. No tools, effects, authority or invented evidence. Return only a JSON object matching this schema: '+json.dumps(request['schema']),
        messages=[dict(role='user',content=request['prompt'])],output_config=dict(effort=metadata['reasoningEffort']))
    if metadata['reasoningEffort']=='none':body['thinking']={'type':'disabled'}
    else:body['thinking']={'type':'enabled'}
    output(dict(event='configuration',resolved=metadata))
    usage={};finished=False;stop=None
    headers={'x-api-key':values['DEEPSEEK_API_KEY'],'anthropic-version':'2023-06-01','content-type':'application/json'}
    with httpx.Client(timeout=httpx.Timeout(5),follow_redirects=False) as client:
        with client.stream('POST',metadata['endpoint'],headers=headers,json=body) as response:
            if response.status_code!=200:raise ModelTransportError('DeepSeek HTTP status '+str(response.status_code))
            for line in response.iter_lines():
                if not line.startswith('data:'):continue
                payload=line[5:].strip()
                if payload=='[DONE]':continue
                event=strict_json(payload);kind=event.get('type')
                if kind=='error':raise ModelTransportError('DeepSeek stream error')
                if kind=='message_start':
                    m=event['message'];usage.update(m.get('usage',{}))
                    if m.get('model')!=metadata['model']:raise ModelTransportError('DeepSeek resolved model differs from configuration')
                elif kind=='content_block_start':
                    block=event['content_block']
                    if block['type'] not in ('text','thinking'):raise ModelTransportError('unexpected DeepSeek tool/content block')
                elif kind=='content_block_delta':
                    delta=event['delta'];channel=delta['type']
                    if channel=='signature_delta':continue
                    if channel not in ('text_delta','thinking_delta'):raise ModelTransportError('unsupported DeepSeek content delta')
                    content=delta.get('text' if channel=='text_delta' else 'thinking','')
                    if content:
                        item=dict(event='content',channel=channel,bytes=len(content.encode()),at=time.monotonic())
                        if channel=='text_delta':item['text']=content
                        else:item['digest']='sha256:'+hashlib.sha256(content.encode()).hexdigest()
                        output(item)
                elif kind=='message_delta':
                    usage.update(event.get('usage',{}));stop=event.get('delta',{}).get('stop_reason',stop)
                elif kind=='message_stop':finished=True;break
    if not finished or stop!='end_turn':
        output(dict(event='completion-status',finished=finished,stop_reason=stop,usage=usage))
        raise ModelTransportError('DeepSeek response incomplete or truncated; stop_reason='+str(stop))
    output(dict(event='completed',usage=usage,resolved=metadata))


class DeepSeekInference:
    def __init__(self,directory,*,env_file):
        self.directory=Path(directory);self.directory.mkdir(parents=True,exist_ok=True)
        self.env_file=str(Path(env_file).expanduser().resolve())
        _,self.resolved=configuration(self.env_file)

    def infer(self,prompt,schema):
        started=last=time.monotonic();first=None;max_gap=0;parts=[];completed=None
        with (self.directory/'events.jsonl').open('w') as journal, (self.directory/'stderr.log').open('wb') as errors:
            child=subprocess.Popen([sys.executable,str(Path(__file__).resolve()),'--worker'],stdin=subprocess.PIPE,
                stdout=subprocess.PIPE,stderr=errors,start_new_session=True)
            child.stdin.write(json.dumps(dict(env_file=self.env_file,prompt=prompt,schema=schema)).encode()+b'\n');child.stdin.close()
            selector=selectors.DefaultSelector();selector.register(child.stdout,selectors.EVENT_READ);buffer=b''
            try:
                while completed is None:
                    now=time.monotonic();remaining=min(180-(now-started),5-(now-last))
                    if remaining<=0:raise ModelTransportError('DeepSeek progress/total deadline exceeded; phase='+('await-first-content' if first is None else 'streaming')+'; idle_seconds='+str(round(now-last,3)))
                    ready=selector.select(remaining)
                    if not ready:continue
                    data=os.read(child.stdout.fileno(),65536)
                    if not data:raise ModelTransportError('DeepSeek worker closed before completion')
                    buffer+=data
                    while b'\n' in buffer:
                        line,buffer=buffer.split(b'\n',1);event=strict_json(line)
                        journal.write(json.dumps(dict(received_at=time.monotonic(),event=event))+'\n');journal.flush()
                        if event['event']=='error':raise ModelTransportError(event['error'])
                        if event['event']=='content' and event['bytes']>0:
                            stamp=event['at']
                            if stamp-last>5 or time.monotonic()-started>180:raise ModelTransportError('DeepSeek content arrived after deadline')
                            max_gap=max(max_gap,stamp-last);last=stamp
                            if first is None:first=stamp-started
                            if event['channel']=='text_delta':parts.append(event['text'])
                            print('MODEL-CONTENT',event['channel'],'bytes='+str(event['bytes']),flush=True)
                        if event['event']=='completed':completed=event
                if child.wait(timeout=1)!=0:raise ModelTransportError('DeepSeek worker failed')
            finally:
                selector.close();child.stdout.close()
                if child.poll() is None:os.killpg(child.pid,signal.SIGKILL);child.wait(timeout=2)
        text=''.join(parts);strict_json(text)
        raw=completed['usage'];usage=dict(last=dict(inputTokens=raw.get('input_tokens',0),
            cachedInputTokens=raw.get('cache_read_input_tokens',0),outputTokens=raw.get('output_tokens',0),
            totalTokens=raw.get('input_tokens',0)+raw.get('output_tokens',0)),provider=raw)
        return dict(text=text,elapsed_seconds=time.monotonic()-started,first_content_seconds=first,
            max_content_gap_seconds=max_gap,usage=usage,resolved=completed['resolved'],
            prompt_digest='sha256:'+hashlib.sha256(prompt.encode()).hexdigest())

    def close(self):pass


if __name__=='__main__':
    try:worker(strict_json(sys.stdin.readline()))
    except Exception as error:
        # Never serialize request headers, credentials, response bodies or exception repr.
        safe=str(error) if isinstance(error,(ModelTransportError,ValueError)) else type(error).__name__
        output(dict(event='error',error=safe));raise SystemExit(1)
